cmake_minimum_required(VERSION 3.18)
project(evtq VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -O3)

if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    message(FATAL_ERROR "Requires x86_64. Detected: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
include(CheckCXXCompilerFlag)
check_cxx_compiler_flag("-mavx2" HAS_AVX2)
check_cxx_compiler_flag("-mfma" HAS_FMA)
if(NOT HAS_AVX2 OR NOT HAS_FMA)
    message(FATAL_ERROR "Requires AVX2 and FMA.")
endif()
add_compile_options(-march=native -mavx2 -mfma)

find_package(OpenMP REQUIRED)
find_package(Python COMPONENTS Interpreter Development REQUIRED)
set(PYBIND11_FINDPYTHON ON)
find_package(pybind11 REQUIRED)

pybind11_add_module(_core src/bindings.cpp)
target_include_directories(_core PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(_core PRIVATE OpenMP::OpenMP_CXX)
