cmake_minimum_required(VERSION 3.25)
project(evtq VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    include(CheckCXXCompilerFlag)
    check_cxx_compiler_flag("-mavx2" HAS_AVX2)
    check_cxx_compiler_flag("-mfma" HAS_FMA)
    if(NOT HAS_AVX2 OR NOT HAS_FMA)
        message(FATAL_ERROR "Requires AVX2 and FMA on x86_64.")
    endif()
    add_compile_options(-Wall -Wextra -march=native -mavx2 -mfma)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
    find_path(SIMDE_INCLUDE_DIR simde/x86/avx2.h HINTS /opt/homebrew/include /usr/include /usr/local/include)
    if(NOT SIMDE_INCLUDE_DIR)
        message(FATAL_ERROR "SIMDe not found. Install with: brew install simde")
    endif()
    add_compile_options(-Wall -Wextra -march=native)
    add_compile_definitions(SIMDE_ENABLE_NATIVE_ALIASES)
    include_directories(SYSTEM ${SIMDE_INCLUDE_DIR})
    if(APPLE)
        execute_process(COMMAND brew --prefix libomp OUTPUT_VARIABLE LIBOMP_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE)
        set(OpenMP_ROOT ${LIBOMP_PREFIX})
    endif()
else()
    message(FATAL_ERROR "Unsupported architecture: ${CMAKE_SYSTEM_PROCESSOR}")
endif()

if(CMAKE_BUILD_TYPE STREQUAL "Release" OR NOT CMAKE_BUILD_TYPE)
    add_compile_options(-O3)
elseif(CMAKE_BUILD_TYPE STREQUAL "Debug")
    add_compile_options(-O0 -g)
elseif(CMAKE_BUILD_TYPE STREQUAL "ASAN")
    add_compile_options(-O1 -g -fsanitize=address)
    add_link_options(-fsanitize=address)
elseif(CMAKE_BUILD_TYPE STREQUAL "UBSAN")
    add_compile_options(-O1 -g -fsanitize=undefined)
    add_link_options(-fsanitize=undefined)
elseif(CMAKE_BUILD_TYPE STREQUAL "TSAN")
    add_compile_options(-O1 -g -fsanitize=thread)
    add_link_options(-fsanitize=thread)
endif()

find_package(OpenMP REQUIRED)
find_package(Python COMPONENTS Interpreter Development REQUIRED)
set(PYBIND11_FINDPYTHON ON)
find_package(pybind11 REQUIRED)

pybind11_add_module(evtq src/bindings.cpp)
target_include_directories(evtq PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(evtq PRIVATE OpenMP::OpenMP_CXX)
